Assignment 9: Create First Recipe
Create:
meta-training/
└── recipes-example/
    └── hello/
        ├── hello.bb
        └── hello.c

hello.bb:
SUMMARY = "Hello World Application"
LICENSE = "MIT"
SRC_URI = "file://hello.c"

do_compile() {
    ${CC} hello.c -o hello
}

do_install() {
    install -d ${D}${bindir}
    install -m 0755 hello ${D}${bindir}
}
